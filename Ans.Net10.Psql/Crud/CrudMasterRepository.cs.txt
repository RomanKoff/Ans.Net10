using Ans.Net10.Common.Crud;
using Microsoft.EntityFrameworkCore;

namespace Ans.Net10.Psql.Crud
{

	/// <summary>
	/// Репозиторий для работы с главными (Master) сущностями PostgreSQL.
	/// </summary>
	/// <typeparam name="T">Тип доменной сущности, реализующей IMasterEntity.</typeparam>
	public class CrudMasterRepository<T>
		: CrudRepository<T>,
		ICrudMasterRepository<T>
		where T : class, IMasterEntity, new()
	{

		public CrudMasterRepository(
			DbContext dbContext)
			: base(dbContext)
		{
		}


		public virtual T GetNew()
		{
			return new T();
		}


		public virtual int GetItemsCount()
		{
			return DbSet.Count();
		}

	}

}
